union Mode
  case light constant
  case dark constant

record Colors
  accent: string required
  surface: string required

environment colorMode: Mode =
  Mode.light

environment highContrast: boolean =
  false

environment locale: string =
  "en-US"

environment color: Colors =
  if env.colorMode is
    Mode.light =>
      <Colors accent="blue" surface="white" />
    Mode.dark =>
      <Colors accent="navy" surface="black" />

environment tone: string =
  if env.highContrast then
    if env.colorMode is
      Mode.light =>
        "ink"
      Mode.dark =>
        "chalk"
  else
    "soft"

environment space: int =
  4

environment gap: int =
  0

  1

  0

  ["red", "green", "blue"]

  (env.space mul 3)

  <env.Update
    env.scale=(scale add 1)
    env.ring=(scale mul 10)
  />

function darkAccent(): string =
  provide env.colorMode=Mode.dark {
    env.color.accent
  }

function lightAccent(): string =
  provide env.colorMode=Mode.light {
    env.color.accent
  }

function frenchAccent(): string =
  provide env.locale="fr-FR" {
    env.color.accent
  }

function redAccent(): string =
  provide env.color.accent="red" {

  }

  provide env.colorMode=Mode.dark {
    env.tone
  }
